\def\wtdatadir{figures/data}

\definecolor{wtResL}{HTML}{E76F51}
\definecolor{wtResR}{HTML}{8F2D14}
\definecolor{wtIn}{HTML}{1D4E89}
\definecolor{wtOut}{HTML}{2A9D8F}

\pgfplotsset{
  wtpanel/.style={
    scale only axis,
    width=0.415\textwidth, height=4.1cm,
    xmin=0, xmax=190, xtick={0,30,...,180},
    ymin=18, ymax=54,  ytick={20,30,...,50},
    grid=major, grid style={dashed, black!15},
    axis x line*=bottom, axis y line*=left,
    tick align=outside, tick label style={font=\small},
    label style={font=\small},
    title style={font=\small, yshift=-3pt},
    every axis plot/.append style={line width=0.8pt, no markers},
  },
  wtL/.style={wtResL},
  wtR/.style={wtResR, densely dashed},
  wtI/.style={wtIn},
  wtO/.style={wtOut, densely dashed},
}
